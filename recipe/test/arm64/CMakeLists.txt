cmake_minimum_required(VERSION 3.19)
project(hdf5_arm64_consumers LANGUAGES C CXX Fortran)
find_package(HDF5 CONFIG REQUIRED COMPONENTS C CXX Fortran shared)
foreach(feature BUILD_FORTRAN BUILD_CPP_LIB BUILD_HL_LIB ENABLE_SZIP_SUPPORT ENABLE_ROS3_VFD)
  if(NOT HDF5_${feature})
    message(FATAL_ERROR "Required HDF5 feature missing: ${feature}")
  endif()
endforeach()
add_executable(hdf5_features features.c)
target_link_libraries(hdf5_features PRIVATE hdf5::hdf5-shared)
add_executable(hdf5_c ../h5_cmprss.c)
target_link_libraries(hdf5_c PRIVATE hdf5::hdf5-shared)
foreach(feature BUILD_FORTRAN BUILD_CPP_LIB BUILD_HL_LIB ENABLE_SZIP_SUPPORT ENABLE_ROS3_VFD)
  if(NOT HDF5_${feature})
    message(FATAL_ERROR "Required HDF5 feature missing: ${feature}")
  endif()
endforeach()
add_executable(hdf5_features features.c)
target_link_libraries(hdf5_features PRIVATE hdf5::hdf5-shared)
add_executable(hdf5_cpp ../h5tutr_cmprss.cpp)
target_link_libraries(hdf5_cpp PRIVATE hdf5::hdf5_cpp-shared)
add_executable(hdf5_fortran ../h5_cmprss.f90)
target_link_libraries(hdf5_fortran PRIVATE hdf5::hdf5_fortran-shared)
add_executable(hdf5_fortran2003 ../compound_fortran2003.f90)
target_link_libraries(hdf5_fortran2003 PRIVATE hdf5::hdf5_fortran-shared)
enable_testing()
foreach(consumer hdf5_features hdf5_c hdf5_cpp hdf5_fortran hdf5_fortran2003)
  add_test(NAME ${consumer} COMMAND ${consumer})
endforeach()
